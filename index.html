function generateSpecificRecommendations(status, age, profession, preferredFields) { let recs = { courses: [], degreeTrack: "", whatToAsk: [] }; // 1. קביעת קורסי AI ממוקדים לפי תחומי העניין והמקצוע if (preferredFields.includes('הייטק ותוכנה')) { recs.courses.push("פיתוח יישומי AI ואינטגרציית LLM (Prompt Engineering & APIs)"); recs.courses.push("אוטומציה עסקית וכלי No-Code/Low-Code עם AI"); } if (preferredFields.includes('כלכלה ופיננסים')) { recs.courses.push("ניתוח נתונים מתקדם וניבוי פיננסי באמצעות AI ו-Python"); recs.courses.push("מודלים של AI בקבלת החלטות פיננסיות"); } if (preferredFields.includes('ניהול ועסקים')) { recs.courses.push("ניהול מוצר וטרנספורמציה דיגיטלית בעידן ה-AI (AI Product Management)"); } // ברירת מחדל לקורס AI בסיסי אם לא נבחר תחום ספציפי if (recs.courses.length === 0) { recs.courses.push("יישום כלי GenAI לייעול תהליכי עבודה ופרודוקטיביות"); } // 2. קביעת מסלול אקדמי / מקצועי (תואר שני / הנדסאים / תעודה) if (status === 'academic_degree') { recs.degreeTrack = "תואר שני (M.Sc / MBA) בניהול טכנולוגיה ונתונים או תוכנית הסבה אקדמית מרוכזת (מבלי להתחיל תואר ראשון מאפס)."; } else if (age >= 30) { recs.degreeTrack = "מסלול הנדסאים (הנדסאי תוכנה / תעשייה וניהול) - לימודי ערב מותאמים לעובדים או מכינת בוגרים (30+)."; } else { recs.degreeTrack = "תואר ראשון מדעי/טכנולוגי (B.Sc / B.A במערכות מידע) או מסלול הנדסאים מואץ."; } // 3. מה בדיוק לבקש/לשאול במוסד הלימודים recs.whatToAsk = [ `"אני מבקש להירשם לקורס ממוקד בנושא: ${recs.courses[0]}"`, `"מהם תנאי הפטור או ההכרה בלימודים קודמים עבור מסלול: ${recs.degreeTrack.split('-')[0]}?"`, `"האם התוכנית כוללת סדנאות מעשיות בשימוש בכלי AI עדכניים ופרויקט גמר יישומי לתעשייה?"` ]; return recs; } // שילוב התוצאה המפורטת ברנדור התשובה function renderEnhancedResults(data) { const recs = generateSpecificRecommendations(data.status, data.age, data.profession, data.fields); const pivotContainer = document.getElementById('ai-pivot-recommendation'); if (pivotContainer) { pivotContainer.innerHTML = `

🤖 תוכנית פעולה מומלצת וממוקדת מול מוסד הלימודים

🎓 מסלול הלימודים הראשי המומלץ:

${recs.degreeTrack}

📘 קורסי AI ממוקדים שמומלץ לדרוש:
    ${recs.courses.map(c => `
  • ${c}
    • `).join('')}
💬 מה בדיוק לבקש מיועץ הלימודים במוסד:
    ${recs.whatToAsk.map(q => `
  • ${q}
    • `).join('')}
`; } }
